void RobotLine::loop() //ubačeno 03102026
{
   // =====================================================
    // STANJA
    // =====================================================

    static bool startDone = false;
    static int lastRightDist = 180;

    // Kamera
    static bool cameraArmed = false;
    static int lowBlueCount = 0;
    static int blueCount = 0;

    // Zavoj
    static bool turningLeft = false;
    static int turnPhase = 0;
    static unsigned long phaseStart = 0;

    // Ponovno aktiviranje kamere nakon zavoja
    static unsigned long rearmStart = 0;
    static int clearBlueCount = 0;
	static unsigned long turnExitTime = 0;  //kada izađe iz faze tri da ne ide prema zidu
	static bool postTurnActive = false;
	static double blueTrigger = 0.005;
	
	// Pronalaženje desnog zida
    static int wallFoundCount = 0;

    // Recovery u fazi 3
    static bool wallRecovery = false;
    static unsigned long wallRecoveryStart = 0;

	static int rightAcquireCount = 0;
	static bool rightWallAcquired = false;

	static int lastAlignDist = 0;
	static int alignStableCount = 0;
	static int alignLostCount = 0;
	static bool phase3Aligning = false;
	static unsigned long phase3AlignStart = 0;

	static bool afterRecoverySearch = false;

	static int lostRightCount = 0;
	
	static bool buildPrinted = false;

	
	if (!buildPrinted)
{
    	print(">>> BUILD 2026-10-07-2140 <<<\n\r");
    	buildPrinted = true;
}


    // =====================================================
    // START
    // =====================================================

    if (!startDone)
    {
        print(">>> MOJA LOOP PETLJA START <<<\n\r");
		
		motorsOn = true;

        mrm_servo->write(59, 1);

        delay(500);

        startDone = true;
    }


    // =====================================================
    // DESNI LIDAR
    // =====================================================

    int rightRaw = mrm_lid_can_b->distance(1);
    int rightDist = rightRaw;

    if (rightRaw > 0 && rightRaw < 700)
    {
        lastRightDist = rightRaw;
    }
    else
    {
        rightDist = lastRightDist;
    }


    // =====================================================
    // KAMERA - DETEKCIJA PLAVE
    // =====================================================

    double blueRatio = 0.0;
    bool cameraFrameOk = false;

    camera->imageGet();

    if (camera->newFrameAvailable && !camera->frame->empty())
    {
        cv::Mat frame = *camera->frame;

        int w = frame.cols;
        int h = frame.rows;

        int roiY = (2 * h) / 3;
        int roiH = h - roiY;

        cv::Rect roi(0, roiY, w, roiH);

        cv::Mat part = frame(roi);

        cv::Mat hsv;
        cv::Mat blueMask;

        cv::cvtColor(part, hsv, cv::COLOR_RGB2HSV);

        cv::inRange(
            hsv,
            cv::Scalar(90, 80, 60),
            cv::Scalar(130, 255, 255),
            blueMask
        );

        int bluePixels = cv::countNonZero(blueMask);
        int totalPixels = blueMask.rows * blueMask.cols;

        if (totalPixels > 0)
        {
            blueRatio =
                (double)bluePixels /
                (double)totalPixels;
        }

        cameraFrameOk = true;

        camera->newFrameAvailable = false;
    }


    // =====================================================
    // PRVO ARMIRANJE KAMERE
    // =====================================================

    if (
        cameraFrameOk &&
        !cameraArmed &&
        !turningLeft &&
        rearmStart == 0
    )
    {
        if (blueRatio <= 0.003)  
        {
            lowBlueCount++;
        }
        else
        {
            lowBlueCount = 0;
        }

        if (lowBlueCount >= 5)
        {
            cameraArmed = true;

            lowBlueCount = 0;
            blueCount = 0;

            print(">>> CAMERA ARMED <<<\n\r");
        }
    }


    // =====================================================
    // PONOVNO ARMIRANJE NAKON ZAVOJA
    // =====================================================

    if (
        cameraFrameOk &&
        !cameraArmed &&
        !turningLeft &&
        rearmStart > 0
    )
    {
         if (millis() - rearmStart >= 2000)
    {
        if (blueRatio <= 0.002)
        {
            clearBlueCount++;
        }
        else
        {
            clearBlueCount = 0;
        }

        if (clearBlueCount >= 5)
        {
            cameraArmed = true;

            clearBlueCount = 0;
            blueCount = 0;
            lowBlueCount = 0;

            rearmStart = 0;

            print(">>> CAMERA REARMED <<<\n\r");
        }
    }
    }


    // =====================================================
    // DETEKCIJA PLAVE - START ZAVOJA
    // =====================================================

    if (
        cameraFrameOk &&
        cameraArmed &&
        !turningLeft &&
        rearmStart == 0
    )
    {
        if (blueRatio >= blueTrigger)  //ČITANJE NIJANSE PLAVE OVISNO O KOLIČINI SVJETLOSTI, GLEDAJTE LOG BLUE, 11,7V
        {
            blueCount++;
        }
        else
        {
            blueCount = 0;
        }

        if (blueCount >= 2)  //UOČAVANJE NA VRIJEME PLAVE BOJE, OVISNO O KOLIČINI SVJETLOSTI!!
        {
            turningLeft = true;

            turnPhase = 1;
            phaseStart = millis();

            cameraArmed = false;

            blueCount = 0;
            lowBlueCount = 0;
            clearBlueCount = 0;

            print(
                ">>> PLAVA CRTA - START LIJEVOG ZAVOJA <<<\n\r"
            );
        }
    }


    // =====================================================
    // LIJEVI ZAVOJ - STATE MACHINE
    // =====================================================

    if (turningLeft)
    {
        // =================================================
        // FAZA 1
        // =================================================

        if (turnPhase == 1)
        {
            print("PHASE1 SERVO=119\n\r");
			
			mrm_servo->write(119, 1);

            go(35, 35);

            if (millis() - phaseStart >= 3750)
            {
                turnPhase = 2;
                phaseStart = millis();

                mrm_servo->write(59, 1);

                print(
                    ">>> ZAVOJ FAZA 2 - UNATRAG <<<\n\r"
                );
            }
        }


        // =================================================
        // FAZA 2
        // =================================================

        else if (turnPhase == 2)
        {
            unsigned long phaseTime = millis() - phaseStart;

    // 1. STANI i unaprijed zakreni kotače za rikverc
    if (phaseTime < 400)
    {
        go(0, 0);
        mrm_servo->write(35, 1); //promjena 23:00
    }

    // 2. RIKVERC sa zakrenutim kotačima
    else if (phaseTime < 1100)
    {
        mrm_servo->write(25, 1);
        go(-35, -35);
    }

    // 3. STANI i pripremi kotače jako ulijevo za fazu 3
    else if (phaseTime < 1400)
    {
        go(0, 0);
        mrm_servo->write(119, 1);
    }

    // 4. FAZA 3
    else
    {
        turnPhase = 3;
        phaseStart = millis();

        wallFoundCount = 0;
        wallRecovery = false;

		rightAcquireCount = 0;
		rightWallAcquired = false;

		print(
                    ">>> ZAVOJ FAZA 3 - TRAZIM DESNI ZID <<<\n\r"
                );
            }
        }


        // =================================================
        // FAZA 3
        // =================================================

        else if (turnPhase == 3)
        {
           unsigned long phaseTime = millis() - phaseStart;

		
    // -------------------------------------------------
    // 1. PRVO AKTIVNO TRAŽI DESNI ZID
    // -------------------------------------------------
    if (!rightWallAcquired)
    {
       if (afterRecoverySearch)
    {
        // Nakon recoveryja smo se već odmaknuli od zida.
        // Ne nastavljaj lijevo, nego lagano traži zid DESNO.
        if (rightRaw > 0 && rightRaw < 180)
            mrm_servo->write(90, 1);   // ipak preblizu -> lijevo
        else
            mrm_servo->write(40, 1);   // umjereno desno

        go(35, 35);
    }
    else
    {
        // Prvi ulazak u fazu 3 ostaje kakav jest jer dobro radi
        mrm_servo->write(95, 1);
        go(40, 40);
    }

    if (rightRaw > 180 && rightRaw < 700)
    {
        rightAcquireCount++;
    }
    else
    {
        rightAcquireCount = 0;
    }

    if (rightAcquireCount >= 2)
    {
        rightWallAcquired = true;
        rightAcquireCount = 0;
        afterRecoverySearch = false;

        phaseStart = millis();
        wallFoundCount = 0;
        wallRecovery = false;

        print(">>> DESNI ZID UHVACEN - NASTAVLJAM FAZU 3 <<<\n\r");
    }
    }

    // -------------------------------------------------
    // 2. DESNI ZID JE PRONAĐEN -> DOSADAŠNJA LOGIKA
    // -------------------------------------------------
    else
    {
        phaseTime = millis() - phaseStart;

		// Ako smo nakon hvatanja zida izgubili desni lidar,
		// nemoj nastaviti skretati ulijevo.
		if (!wallRecovery && !phase3Aligning)
		{
   	 		if (rightRaw <= 0 || rightRaw >= 700)
        		lostRightCount++;
    		else
        		lostRightCount = 0;
		}

		if (lostRightCount >= 4)
	{
   			rightWallAcquired = false;
    		rightAcquireCount = 0;
    		wallFoundCount = 0;
    		lostRightCount = 0;

    		afterRecoverySearch = true;

    		phaseStart = millis();

    		mrm_servo->write(40, 1);
    		go(35, 35);

    	print(">>> IZGUBLJEN DESNI ZID - TRAZIM PONOVNO <<<\n\r");
	}

        // Ako smo preblizu desnom zidu -> recovery
    else if (!wallRecovery &&
            rightRaw > 0 &&
            rightRaw < 140)
        {
            wallRecovery = true;
            wallRecoveryStart = millis();
            wallFoundCount = 0;

			// Recovery mijenja geometriju robota,
    		// zato staro poravnanje više ne vrijedi
    		phase3Aligning = false;
    		alignStableCount = 0;
    		lastAlignDist = 0;

            print(">>> PREBLIZU ZIDU - RECOVERY <<<\n\r");
        }

        if (wallRecovery)
        {
            unsigned long recoveryTime =
                millis() - wallRecoveryStart;

            // RIKVERC
            if (recoveryTime < 800)
            {
                mrm_servo->write(25, 1);
                go(-35, -35);
            }

            // STANI I ZAKRENI LIJEVO
            else if (recoveryTime < 1050)
            {
                go(0, 0);
                mrm_servo->write(119, 1);
            }

            // KRENI NAPRIJED LIJEVO
            else if (recoveryTime < 1750)
            {
                mrm_servo->write(119, 1);
                go(30, 30);
            }

            // GOTOV RECOVERY
            else
            {
                
    			wallRecovery = false;
    			wallFoundCount = 0;

    			// Ako nas je recovery vratio na dobru udaljenost,
    			// zavoj je gotov - izađi iz faze 3
    		if (rightDist >= 150 && rightDist <= 260)
    	{
        		phase3Aligning = false;

        		turningLeft = false;
        		turnPhase = 0;

        	rightWallAcquired = false;
        	rightAcquireCount = 0;

        	alignStableCount = 0;
        	alignLostCount = 0;
        	lostRightCount = 0;

        	afterRecoverySearch = false;

        	blueCount = 0;
        	lowBlueCount = 0;
        	clearBlueCount = 0;

        	cameraArmed = false;
       		 rearmStart = millis();

        	turnExitTime = millis();
        	postTurnActive = true;

        	mrm_servo->write(59, 1);

        	print(">>> RECOVERY GOTOV - PRATIM DESNI ZID <<<\n\r");
    }
    else
    {
        rightWallAcquired = false;
        rightAcquireCount = 0;

        afterRecoverySearch = true;

        phaseStart = millis();
    }
            }
        }

        // Kratki početak nakon hvatanja zida
        else if (phaseTime < 500)
        {
            mrm_servo->write(115, 1);
            go(30, 30);
        }

        // Ako smo još preblizu ciljnom području
        else if (rightRaw >= 100 &&
                 rightRaw < 180)
        {
            mrm_servo->write(115, 1);
            go(30, 30);

            wallFoundCount = 0;
        }

        // Normalno traženje položaja uz zid
        else
        {
        // -------------------------------------------------
    	// ZAVRŠNO PORAVNAVANJE
    	// -------------------------------------------------
	if (phase3Aligning)
{
    mrm_servo->write(119, 1);
    go(30, 30);

    if (rightRaw <= 0 || rightRaw >= 700)
    {
        alignLostCount++;
    }
    else
    {
        alignLostCount = 0;
    }

    if (alignLostCount >= 4)
    {
        // NISMO poravnati - izgubili smo desni zid
        phase3Aligning = false;

        rightWallAcquired = false;
        rightAcquireCount = 0;
        wallFoundCount = 0;

        alignStableCount = 0;
        alignLostCount = 0;

        afterRecoverySearch = true;
        phaseStart = millis();

        print(">>> IZGUBLJEN ZID TIJEKOM PORAVNAVANJA - TRAZIM PONOVNO <<<\n\r");
    }

    // Prvih 800 ms OBAVEZNO nastavi poravnavanje
    else if (millis() - phase3AlignStart >= 700)
    {
        if (rightRaw > 0 && rightRaw < 800)
        {
            // Dovoljno smo blizu desnom zidu
            // i više mu se značajno ne približavamo
            if (rightDist >= 160 &&
                rightDist <= 260 &&
                rightDist >= lastAlignDist - 5)
            {
                alignStableCount++;
            }
            else
            {
                alignStableCount = 0;
            }

            lastAlignDist = rightDist;
        }
        else
        {
            alignStableCount = 0;
        }

        // Tek nakon stabilnih očitanja izlazimo iz zavoja
        if (alignStableCount >= 3 ||
            (millis() - phase3AlignStart >= 2900 &&
             alignStableCount >= 2))
        {
            phase3Aligning = false;

            turningLeft = false;
            turnPhase = 0;

            wallFoundCount = 0;
            blueCount = 0;
            lowBlueCount = 0;
            clearBlueCount = 0;

            wallRecovery = false;

            cameraArmed = false;
            rearmStart = millis();

            turnExitTime = millis();
            postTurnActive = true;

            mrm_servo->write(59, 1);

            print(">>> PORAVNAT - PRATIM DESNI ZID <<<\n\r");
        }
    }
}
    
	
    // -------------------------------------------------
    // JOŠ NISMO U ZAVRŠNOM PORAVNAVANJU
    // -------------------------------------------------
    else
    {
        mrm_servo->write(115, 1);
        go(30, 30);

        if (rightRaw >= 180 &&
            rightRaw <= 500)
        {
            wallFoundCount++;
        }
        else
        {
            wallFoundCount = 0;
        }

        // 3 uzastopna dobra očitanja -> počni poravnanje
        if (wallFoundCount >= 3)
        {
            phase3Aligning = true;
            phase3AlignStart = millis();

            lastAlignDist = rightDist;
            alignStableCount = 0;

			alignLostCount = 0;
            wallFoundCount = 0;

            print(">>> DESNI ZID PRONADJEN - PORAVNAVAM <<<\n\r");
        }
    }
        }
    }
        }

}
    // =====================================================
    // NORMALNO PRAĆENJE DESNOG ZIDA
    // =====================================================

    else
    {
        motorsOn = true;

if (postTurnActive && millis() - turnExitTime < 700)
{
    if (rightDist < 180)
        mrm_servo->write(90, 1);
    else
        mrm_servo->write(59, 1);

    go(35, 35);
}
else
{
    postTurnActive = false;

    if (rightDist < 160)
    {
        mrm_servo->write(90, 1);
    }
    else if (rightDist <= 210)
    {
        mrm_servo->write(59, 1);
    }
    else
    {
        mrm_servo->write(25, 1);
    }

    go(30, 30);
}
    }


    // =====================================================
    // ISPIS
    // =====================================================

    //print(">>> VERZIJA 14:35 - PHASE3 35 <<<\n\r");
	print(
        "RAW:%i R:%i BLUE:%.6f armed:%i blueCount:%i turn:%i phase:%i\n\r",
        rightRaw,
        rightDist,
        blueRatio,
        cameraArmed,
        blueCount,
        turningLeft,
        turnPhase
    );

    delay(200);
}



void RobotLine::loop0()
{
	threadClearToRun = false;
	uint8_t data[8];

	uint8_t errorCode = 0;
	while  (!userBreak()){
		data[0] = 0xFF;
		CANMessage msg(0x180, data, 1);
		uart->write(msg, errorCode, true);
		delay(5);
		CANMessage message[5];
		int8_t last;
		uart->read(message, last, errorCode, true);
	}
	threadClearToRun = true;
	end();
}
